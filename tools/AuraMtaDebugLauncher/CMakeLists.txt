cmake_minimum_required(VERSION 3.20)
project(AuraMtaDebugLauncher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(AuraMtaDebugLauncher Launcher.cpp)
target_include_directories(AuraMtaDebugLauncher PRIVATE ../../include)
target_compile_definitions(AuraMtaDebugLauncher PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
get_filename_component(TRIAGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
target_compile_definitions(AuraMtaDebugLauncher PRIVATE TRIAGE_ROOT=L"${TRIAGE_ROOT}")
target_compile_options(AuraMtaDebugLauncher PRIVATE /W4 /WX /utf-8 /Zi)
target_link_options(AuraMtaDebugLauncher PRIVATE /DEBUG)
target_link_libraries(AuraMtaDebugLauncher PRIVATE dbghelp)
add_executable(DebugFixture Fixture.cpp)
target_compile_options(DebugFixture PRIVATE /W4 /WX /Zi)
target_link_options(DebugFixture PRIVATE /DEBUG)
