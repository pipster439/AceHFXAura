cmake_minimum_required(VERSION 3.20)
project(AuraOwnershipExperiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../src/AuraWorker/CanonicalAuraAbi.h" AURA_ABI_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../docs/research/aura-ownership-callsite-analysis.md" AURA_REVIEW_HASH)
configure_file(ReviewedEvidence.h.in ReviewedEvidence.h @ONLY)
add_library(AuraGateACore STATIC Experiment.cpp Journal.cpp)
target_include_directories(AuraGateACore PUBLIC . ../../include)
target_compile_definitions(AuraGateACore PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(AuraGateACore PRIVATE /W4 /WX /utf-8)
add_executable(AuraOwnershipExperiment Main.cpp NativeAuraApi.cpp)
target_include_directories(AuraOwnershipExperiment PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" ../../src/AuraWorker)
target_link_libraries(AuraOwnershipExperiment PRIVATE AuraGateACore ole32 oleaut32 advapi32 bcrypt version shell32 wtsapi32 windowsapp)
target_compile_options(AuraOwnershipExperiment PRIVATE /W4 /WX /utf-8)
option(AURA_GATE_A_SOFTWARE_TESTS "Build only software fixtures/tests alongside candidate" OFF)
if(AURA_GATE_A_SOFTWARE_TESTS)
  enable_testing()
  add_executable(AuraGateATests ../../tests/AuraOwnershipExperiment.Tests/CoreTests.cpp)
  target_link_libraries(AuraGateATests PRIVATE AuraGateACore)
  target_compile_options(AuraGateATests PRIVATE /W4 /WX /utf-8)
  add_test(NAME aura_gate_a_software COMMAND AuraGateATests)
  add_executable(AuraGateAOwnerFixture ../../tests/AuraOwnershipExperiment.Tests/OwnerFixture.cpp)
  target_link_libraries(AuraGateAOwnerFixture PRIVATE AuraGateACore)
  target_compile_options(AuraGateAOwnerFixture PRIVATE /W4 /WX /utf-8)
endif()
