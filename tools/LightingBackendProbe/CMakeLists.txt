cmake_minimum_required(VERSION 3.20)
project(LightingBackendProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(LightingBackendProbe Probe.cpp MediatorReadOnly.cpp)
target_include_directories(LightingBackendProbe PRIVATE ../../include)
target_compile_definitions(LightingBackendProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(LightingBackendProbe PRIVATE /W4 /WX /utf-8)
target_link_libraries(LightingBackendProbe PRIVATE ole32 oleaut32 advapi32 runtimeobject shell32)
