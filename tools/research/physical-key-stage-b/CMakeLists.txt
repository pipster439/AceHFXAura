cmake_minimum_required(VERSION 3.20)
project(aura_physical_key_stage_b LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(stage_b_offline_tests offline_tests.cpp delivery_tests.cpp)
target_compile_definitions(stage_b_offline_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(stage_b_offline_tests PRIVATE /W4 /WX /utf-8)
endif()
if(WIN32)
  add_executable(stage_b_observer observer_win32.cpp)
  target_compile_definitions(stage_b_observer PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(stage_b_observer PRIVATE user32 hid wtsapi32 bcrypt)
  if(MSVC)
    target_compile_options(stage_b_observer PRIVATE /W4 /WX /utf-8)
  endif()
endif()
enable_testing()
add_test(NAME stage_b_offline COMMAND stage_b_offline_tests)
set_tests_properties(stage_b_offline PROPERTIES TIMEOUT 15)
