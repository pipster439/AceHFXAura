cmake_minimum_required(VERSION 3.20)
project(AuraMtaCrashProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_abi.py"
 "${CMAKE_CURRENT_SOURCE_DIR}/../../src/AuraWorker/CanonicalAuraAbi.h" "${CMAKE_CURRENT_BINARY_DIR}/ProbeAbi.h" COMMAND_ERROR_IS_FATAL ANY)
foreach(target AuraMtaCrashProbe AuraMtaPreflightProbe)
 add_executable(${target} Probe.cpp)
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
 target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /Zi)
 target_link_options(${target} PRIVATE /DEBUG)
 target_link_libraries(${target} PRIVATE ole32 oleaut32 advapi32 runtimeobject)
endforeach()
target_compile_definitions(AuraMtaPreflightProbe PRIVATE RICH_PREFLIGHT=1)
target_link_libraries(AuraMtaPreflightProbe PRIVATE bcrypt version shell32)
