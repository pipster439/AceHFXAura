cmake_minimum_required(VERSION 3.20)
project(AuraWorker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(AuraWorker AuraWorker.cpp)
target_include_directories(AuraWorker PRIVATE ../../include)
target_compile_definitions(AuraWorker PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(AuraWorker PRIVATE /W4 /WX /utf-8)
target_link_libraries(AuraWorker PRIVATE ole32 oleaut32 advapi32)
option(AURA_WORKER_FIXTURES "Build software-only worker fault fixture" OFF)
if(AURA_WORKER_FIXTURES)
  add_executable(AuraWorkerFixture ../../tests/AsusPlatform.Tests/AuraWorkerFixture.cpp)
  target_include_directories(AuraWorkerFixture PRIVATE . ../../include)
  target_compile_definitions(AuraWorkerFixture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(AuraWorkerFixture PRIVATE /W4 /WX /utf-8)
endif()
